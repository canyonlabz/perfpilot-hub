# PerfPilot Agents - Python dependencies
#
# Mirror of pyproject.toml dependencies for environments that prefer pip-style
# requirements files (Docker, virtual envs, CI). Floor versions only; pinning
# strategy matches the rest of the suite (e.g. perfmemory-mcp/requirements.txt).
#
# Starlette pinned to >=1.2.1 to mitigate CVE-2026-48710 (host header injection
# / authentication bypass).

# AG2 framework. `[ag-ui]` extra pulls in `ag-ui-protocol` (pinned explicitly
# below) so AG2 can expose the AG-UI endpoint consumed by CopilotKit React.
# `[openai]` brings the OpenAI-backed typed-config stack (also used for Azure
# OpenAI); `[anthropic]` and `[ollama]` bring the Claude and Ollama typed-config
# stacks used by the per-vendor adapters in services/model_adapters/. Pinned
# ==1.3.2 for migration stabilization; relax to >=1.3.2,<1.4 once it lands.
# NOTE: the `ag2[mcp]` extra is intentionally NOT used - it requires mcp>=2.3,
# which conflicts with fastmcp 3.x (mcp<2.0).
ag2[ag-ui,openai,anthropic,ollama]==1.3.2

# AG-UI wire protocol. Transitive via ag2[ag-ui]; pinned explicitly to lock the
# AG-UI 1.0 contract consumed by CopilotKit 1.88 / @ag-ui/client 1.0.2.
ag-ui-protocol==1.0.0

# MCP client
fastmcp>=3.4.2,<4

# Web servers (A2A on port 8001, AG-UI on port 8002)
fastapi>=0.115.0
uvicorn[standard]>=0.32.0
starlette>=1.2.1

# SSE responses for A2A `tasks/sendSubscribe` and AG-UI event stream
sse-starlette>=2.1.3

# Database (perfagent_state pool)
asyncpg>=0.30.0

# Web-UI token encryption and the AG-UI to MCP identity token
cryptography>=43.0.0
PyJWT>=2.9.0

# Models / validation
pydantic>=2.10.0

# HTTP / async
httpx>=0.28.1

# httpx2 is openai>=3's async transport and AG2's typed-config `http_client`
# type; services/llm_provider.py constructs it directly for the agent LLM path.
httpx2>=2.16.2,<3

# LLM provider SDKs. openai>=3 is required by ag2[openai]; the 3.x line uses
# httpx2 (installed transitively) for its async transport. The TLS-aware client
# in services/llm_provider.py moves to httpx2 in the agent-path refactor.
openai>=3.45.0,<4

# Anthropic (Claude) SDK, pulled by ag2[anthropic] and used by
# services/model_adapters/anthropic.py. AG2 1.3.2 allows >=1.15,<3 and targets
# the 2.x Messages API (which rejects a sampling temperature - the adapter omits
# it). Pinned to the 2.x line the adapter was validated against. The Claude CLI
# is a separate tool and is not a Python dependency.
anthropic>=2.12.0,<3

# Ollama SDK, pulled by ag2[ollama] and used by services/model_adapters/ollama.py
# for the local Ollama agent path (native OllamaConfig, host-based).
ollama>=0.5.2,<1

# Config / utilities
pyyaml>=6.0.2
python-dotenv>=1.2.1

# Token counting (context window tracking in multi-turn tool loop)
tiktoken>=0.12.0
