# PerfPilot Agents - Python dependencies (Docker-scoped mirror)
#
# This file is a Docker-scoped mirror of agent-framework/backend/requirements.txt.
# Keeping the mirror inside docker/ matches the per-image convention already
# used by the FastMCP MCPs (docker/<mcp>/requirements.txt) so the Docker build
# context stays self-contained under docker/agent-backend/.
#
# When bumping a version here, mirror the change in
# agent-framework/backend/requirements.txt (and pyproject.toml) so local /
# non-Docker environments stay in sync.
#
# Starlette pinned to >=1.2.1 to mitigate CVE-2026-48710 (host header
# injection / authentication bypass) - same pin as the gateway image.

# AG2 framework. `[ag-ui]` extra pulls in `ag-ui-protocol` so AG2 can expose
# the AG-UI endpoint at /copilotkit via `ag2.ag_ui.AGUIStream` (consumed by
# CopilotKit React). `[openai]` brings the OpenAI-backed typed-config stack
# (also used for Azure OpenAI); `[anthropic]` and `[ollama]` bring the Claude
# and Ollama typed-config stacks used by services/model_adapters/. Pinned
# ==1.3.2 for migration stabilization; relax to >=1.3.2,<1.4 once it lands.
# NOTE: the `ag2[mcp]` extra is intentionally NOT used - it requires mcp>=2.3,
# which conflicts with fastmcp 3.x (mcp<2.0).
ag2[ag-ui,openai,anthropic,ollama]==1.3.2

# AG-UI wire protocol. Transitive via ag2[ag-ui]; pinned explicitly to lock the
# AG-UI 1.0 contract consumed by CopilotKit 1.88 / @ag-ui/client 1.0.2.
ag-ui-protocol==1.0.0

# MCP client
fastmcp>=3.4.2,<4

# Web servers (A2A on port 8101, AG-UI on port 8102)
fastapi>=0.115.0
uvicorn[standard]>=0.32.0
starlette>=1.2.1

# SSE responses for A2A `tasks/sendSubscribe` and AG-UI event stream
sse-starlette>=2.1.3

# Database (perfagent_state pool)
asyncpg>=0.30.0

# Models / validation
pydantic>=2.10.0

# HTTP / async
httpx>=0.28.1

# httpx2 is openai>=3's async transport and AG2's typed-config `http_client`
# type; services/llm_provider.py constructs it directly for the agent LLM path.
httpx2>=2.16.2,<3

# LLM provider SDKs. openai>=3 is required by ag2[openai]; the 3.x line uses
# httpx2 (installed transitively) for its async transport.
openai>=3.45.0,<4

# Anthropic (Claude) SDK, pulled by ag2[anthropic] and used by
# services/model_adapters/anthropic.py. Pinned to the 2.x line the adapter was
# validated against (AG2 1.3.2 allows >=1.15,<3).
anthropic>=2.12.0,<3

# Ollama SDK, pulled by ag2[ollama] and used by services/model_adapters/ollama.py.
ollama>=0.5.2,<1

# Config / utilities
pyyaml>=6.0.2
python-dotenv>=1.2.1

# Structured logging (DEPLOYMENT_MODE local/cloud). Imported directly by
# utils/logging_config.py; kept declared here even though it currently also
# arrives transitively, so the container never loses it on a dependency bump.
structlog>=24.0.0

# Token counting (context window tracking in multi-turn tool loop)
tiktoken>=0.12.0

# Web-UI user-secret encryption (AES-256-GCM) and MCP context JWT (HS256)
cryptography>=43.0.0
PyJWT>=2.9.0
